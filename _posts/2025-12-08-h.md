<script>
function launchSparkles(x, y, count = 200) {
  const container = document.getElementById("sparkle-container");

  for (let i = 0; i < count; i++) {
    const sparkle = document.createElement("div");
    sparkle.className = "sparkle";

    const angle = Math.random() * 2 * Math.PI;
    const speed = 1 + Math.random() * 3;
    const dx = Math.cos(angle) * speed * 30;
    const dy = Math.sin(angle) * speed * 30;

    sparkle.style.left = x + "px";
    sparkle.style.top = y + "px";

    container.appendChild(sparkle);

    requestAnimationFrame(() => {
      sparkle.style.transition = "transform 800ms ease-out, opacity 800ms ease-out";
      sparkle.style.transform = "translate(" + dx + "px, " + dy + "px) scale(0.3)";
      sparkle.style.opacity = "0";
    });

    setTimeout(() => sparkle.remove(), 1500);
  }
}
</script>
<script>
<!-- Typewriter style -->
function typewrite(element, text, speed = 40) {
  element.innerHTML = "";
  let i = 0;

  function typeNext() {
    if (i < text.length) {
      element.innerHTML += text.charAt(i);
      i++;
      setTimeout(typeNext, speed);
    }
  }

  typeNext();
}
</script>

<script>
document.addEventListener("DOMContentLoaded", () => {
  const box = document.getElementById("case-jour");
  const boxText = document.getElementById("box-text");
  const revealed = document.getElementById("revealed-content");
  const revealedimg = document.getElementById("revealed-img");

  box.addEventListener("click", (e) => {
    // Launch sparkles
    launchSparkles(e.clientX, e.clientY, 25);
    
    if (box.dataset.opened === "true") return;
    box.dataset.opened = "true";

    // Change text inside box
    boxText.style.opacity = "0";
    setTimeout(() => {
      boxText.innerHTML = "🎁 Quelques dilemmes à résoudre";
      boxText.style.opacity = "1";
      box.style.backgroundImage = "none";
    }, 400);

    // Show content below box typed-style
    revealed.style.display = "block";
    typewrite(revealed, revealed.textContent.trim(), 5); // 5ms per letter
    revealedimg.style.display = "block";
  });
});
</script>
