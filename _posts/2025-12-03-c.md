<script>
document.addEventListener("DOMContentLoaded", () => {
  const box = document.getElementById("case-jour");
  const boxText = document.getElementById("box-text");
  const glitter = document.getElementById("glitter");
  const revealed = document.getElementById("revealed-content");

  box.addEventListener("click", () => {
    if (box.dataset.opened === "true") return;
    box.dataset.opened = "true";

    // Glitter effect (simple fade)
    glitter.style.opacity = "1";
    setTimeout(() => glitter.style.opacity = "0", 1000);

    // Change text inside box
    boxText.style.opacity = "0";
    setTimeout(() => {
      boxText.innerHTML = "🎁 Cadeau";
      boxText.style.opacity = "1";
      box.style.backgroundImage = "none"; // fade background
    }, 400);

    // Show content below box
    revealed.style.display = "block";
  });
});
</script>
