<script>
function launchSparkles(x, y, count = 200) {
  const container = document.getElementById("sparkle-container");

  for (let i = 0; i < count; i++) {
    const sparkle = document.createElement("div");
    sparkle.className = "sparkle";

    const angle = Math.random() * 2 * Math.PI;
    const speed = 1 + Math.random() * 3;
    const dx = Math.cos(angle) * speed * 30;
    const dy = Math.sin(angle) * speed * 30;

    sparkle.style.left = x + "px";
    sparkle.style.top = y + "px";

    container.appendChild(sparkle);

    requestAnimationFrame(() => {
      sparkle.style.transition = "transform 800ms ease-out, opacity 800ms ease-out";
      sparkle.style.transform = "translate(" + dx + "px, " + dy + "px) scale(0.3)";
      sparkle.style.opacity = "0";
    });

    setTimeout(() => sparkle.remove(), 1500);
  }
}
</script>
<script>
<!-- Typewriter style -->
function typewrite(element, text, speed = 40) {
  element.innerHTML = "";
  let i = 0;

  function typeNext() {
    if (i < text.length) {
      element.innerHTML += text.charAt(i);
      i++;
      setTimeout(typeNext, speed);
    }
  }

  typeNext();
}
</script>
